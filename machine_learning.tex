\section{Grundkonzepte}
%===============================================================================

\begin{frame}
    \frametitle{Was ist Maschinelles Lernen?}
    \begin{columns}
        \begin{column}{0.55\textwidth}
            \begin{block}{Kernidee}
                Statt Regeln manuell zu programmieren, lernt der Computer die Regeln \textbf{aus Daten}.
            \end{block}
            \vspace{0.4cm}
            \textbf{Traditionelle Programmierung:}\\
            Mensch schreibt Regeln $\rightarrow$ Computer führt aus
            \vspace{0.3cm}
            
            \textbf{Maschinelles Lernen:}\\
            Daten + Beispiele $\rightarrow$ Computer lernt Regeln
        \end{column}
        \begin{column}{0.42\textwidth}
            \begin{exampleblock}{Beispiel: Geldwäsche}
                \textbf{Regelbasiert:}\\
                „Melde alle TX $>$ \texteuro 10.000"
                \vspace{0.2cm}
                
                \textbf{ML-basiert:}\\
                System erkennt verdächtige Muster aus historischen Fällen -- auch \textit{unbekannte}.
            \end{exampleblock}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}
    \frametitle{Warum ML? -- Grenzen regelbasierter Systeme}
    \begin{columns}
        \begin{column}{0.48\textwidth}
            \begin{alertblock}{Probleme mit Regeln}
                \begin{itemize}
                    \item \textbf{Starre Schwellenwerte:} \texteuro 9.999 vs. \texteuro 10.001
                    \item \textbf{Bekannte Muster:} Nur was programmiert wurde
                    \item \textbf{Kombinatorische Explosion:} Tausende Wenn-Dann-Regeln
                    \item \textbf{Wartungsaufwand:} Manuelle Anpassung an neue Typologien
                \end{itemize}
            \end{alertblock}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{block}{Vorteile von ML}
                \begin{itemize}
                    \item \textbf{Weiche Grenzen:} Wahrscheinlichkeiten statt Ja/Nein
                    \item \textbf{Mustererkennung:} Auch unbekannte Muster
                    \item \textbf{Skalierbarkeit:} Tausende Features gleichzeitig
                    \item \textbf{Adaptivität:} Lernt aus neuen Daten
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

                \begin{itemize}

                    \item Ziel: Vorhersage für neue Fälle

            \begin{block}{Unüberwacht (Unsupervised)}
                \begin{itemize}
                    \item Training \textbf{ohne Labels}
                    \item Ziel: Strukturen entdecken
                    \item \textit{Bsp: Kundensegmentierung}
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

    \begin{center}

        \begin{column}{0.48\textwidth}
            \begin{block}{Instanzbasiert}
                \begin{itemize}
                    \item Speichert alle Trainingsdaten
                    \item Vergleicht neue Fälle mit Ähnlichen
                    \item \textit{Bsp: k-Nearest Neighbors}

            \end{block}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{block}{Modellbasiert}
                \begin{itemize}
                    \item Lernt Parameter aus Daten
                    \item Kompaktes Modell, schnelle Vorhersage
                    \item \textit{Bsp: Logistische Regression}
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}

\begin{frame}

    \begin{columns}
        \begin{column}{0.48\textwidth}
            \begin{block}{Random Forest}
                \begin{itemize}
                    \item Viele Decision Trees
                    \item Jeder sieht andere Daten
                    \item \textbf{Mehrheitsentscheidung}
                \end{itemize}
                \vspace{0.2cm}
                \textit{Wie ein Richtersenat: Mehrere unabhängige Meinungen führen zu robusterem Urteil.}
            \end{block}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{block}{Gradient Boosting}
                \begin{itemize}
                    \item Bäume lernen \textbf{sequentiell}
                    \item Jeder korrigiert Fehler des Vorgängers
                    \item State-of-the-Art für Tabellen
                \end{itemize}
                \vspace{0.2cm}
                \textit{Wie iterative Rechtsprechung: Jedes Urteil verfeinert die Auslegung.}
            \end{block}
        \end{column}
    \end{columns}
    \vspace{0.3cm}
    \begin{alertblock}{Praxis}
        XGBoost/LightGBM sind der \textbf{De-facto-Standard} für AML-Anwendungen mit strukturierten Daten.
    \end{alertblock}
\end{frame}

\section{Evaluation \& Metriken}
%===============================================================================

\begin{frame}
    \frametitle{Warum Accuracy nicht ausreicht}
    \begin{columns}
        \begin{column}{0.55\textwidth}
            \textbf{Das Dilemma bei seltenen Ereignissen:}
            \begin{itemize}
                \item Geldwäsche: $<$1\% aller Transaktionen
                \item Ein Modell, das \textit{immer} „unbedenklich" sagt, hat 99\% Trefferquote!
            \end{itemize}
            \vspace{0.3cm}
            \textbf{Bessere Maßstäbe:}
            \begin{itemize}
                \item \textbf{Precision:} Wie viele Alarme sind echte Treffer?
                \item \textbf{Recall:} Wie viele echte Fälle wurden gefunden?
                \item \textbf{F1-Score:} Harmonisches Mittel beider
            \end{itemize}
        \end{column}
        \begin{column}{0.42\textwidth}
            \begin{alertblock}{Abwägung}
                \textbf{Hoher Recall:}\\
                Weniger übersehene Fälle\\
                {\small (regulatorisch wichtig!)}
                \vspace{0.2cm}
                
                \textbf{Hohe Precision:}\\
                Weniger Fehlalarme\\
                {\small (Effizienz)}
            \end{alertblock}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}
    \frametitle{Konfusionsmatrix \& Precision-Recall}
    \begin{center}
        \includegraphics[width=0.9\linewidth]{plots/precision_recall.pdf}
    \end{center}
    \vspace{-0.2cm}
    \[
        \text{Precision} = \frac{TP}{TP + FP} \qquad
        \text{Recall} = \frac{TP}{TP + FN} \qquad
        F_1 = 2 \, \frac{P \, R}{P + R}
    \]
\end{frame}

\begin{frame}
    \frametitle{Train/Validation/Test: Das Daten-Dreieck}
    \begin{columns}
        \begin{column}{0.55\textwidth}
            \begin{enumerate}
                \item \textbf{Training Set (60-70\%)}\\
                      Modell lernt daraus
                \item \textbf{Validation Set (15-20\%)}\\
                      Hyperparameter-Tuning
                \item \textbf{Test Set (15-20\%)}\\
                      Finale, unberührte Evaluation
            \end{enumerate}
            \vspace{0.3cm}
            \begin{alertblock}{Goldene Regel}
                Das Test Set darf \textbf{niemals} für Modellentscheidungen genutzt werden!
            \end{alertblock}
        \end{column}
        \begin{column}{0.42\textwidth}
            \begin{block}{Juristische Analogie}
                Wie bei einer \textbf{Prüfung}:
                \begin{itemize}
                    \item Training = Lernen mit Übungsaufgaben
                    \item Validation = Probeklausur
                    \item Test = Echte Prüfung
                \end{itemize}
                \vspace{0.2cm}
                Wer die Prüfungsfragen vorher kennt, besteht -- aber kann er auch wirklich etwas?
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}
    \frametitle{Cross-Validation: Robuste Schätzung}
    \begin{columns}
        \begin{column}{0.55\textwidth}
            \textbf{K-Fold Cross-Validation:}
            \begin{enumerate}
                \item Teile Daten in $k$ gleiche Teile
                \item Trainiere $k$ mal, jedes Mal mit anderem Test-Teil
                \item Mittle die Ergebnisse
            \end{enumerate}
            \vspace{0.3cm}
            \textbf{Vorteile:}
            \begin{itemize}
                \item Nutzt alle Daten für Training UND Test
                \item Robustere Leistungsschätzung
                \item Erkennt Überanpassung besser
            \end{itemize}
        \end{column}
        \begin{column}{0.42\textwidth}
            \begin{block}{Out-of-Time Validation}
                Für Zeitreihendaten (wie Transaktionen):
                \begin{itemize}
                    \item Training auf alten Daten
                    \item Test auf neuen Daten
                    \item Simuliert Praxis-Einsatz
                \end{itemize}
                \vspace{0.2cm}
                \textit{Besonders wichtig bei AML wegen sich ändernder Typologien!}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}
    \frametitle{Erklärbarkeit: Warum hat das Modell so entschieden?}
    \begin{columns}
        \begin{column}{0.48\textwidth}
            \begin{block}{Feature Importance}
                Welche Eingabevariablen haben den größten Einfluss?
                \begin{itemize}
                    \item Bei Gradient Boosting: eingebaut
                    \item Globale Sicht auf Modell
                \end{itemize}
            \end{block}
            \vspace{0.2cm}
            \begin{block}{SHAP Values}
                Erklärt \textbf{einzelne Vorhersagen}:
                \begin{itemize}
                    \item Warum dieser Score für diesen Kunden?
                    \item Welche Faktoren erhöhen/senken Risiko?
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{alertblock}{Regulatorische Anforderung}
                Erklärbarkeit ist \textbf{keine Option}:
                \begin{itemize}
                    \item Kunde hat Recht auf Erklärung
                    \item Aufsicht muss Modell verstehen
                    \item Interne Revision prüft
                    \item MaRisk/BAIT verlangen Dokumentation
                \end{itemize}
            \end{alertblock}
            \vspace{0.2cm}
            \textit{„Warum wurde mein Konto gesperrt?" muss beantwortbar sein!}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}
    \frametitle{Red Flags bei ML-Implementierungen}
    \begin{columns}
        \begin{column}{0.48\textwidth}
            \begin{alertblock}{Warnsignale}
                \begin{itemize}
                    \item Keine klare \textbf{Dokumentation}
                    \item Test Set wurde für Tuning genutzt
                    \item Nur \textbf{Accuracy} als Metrik
                    \item Kein \textbf{Monitoring} in Produktion
                    \item „Black Box" ohne Erklärbarkeit
                    \item Keine \textbf{Governance}-Struktur
                    \item Modell seit Jahren nicht \textbf{re-trainiert}
                    \item Keine \textbf{Challenger Models}
                \end{itemize}
            \end{alertblock}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{block}{Best Practices}
                \begin{itemize}
                    \item Klare \textbf{Model Risk Management} Policy
                    \item Regelmäßige \textbf{Validierung} (mind. jährlich)
                    \item \textbf{Challenger Models} zum Vergleich
                    \item \textbf{Human-in-the-Loop} bei kritischen Entscheidungen
                    \item Dokumentierte \textbf{Feature-Auswahl}
                    \item \textbf{Audit Trail} aller Änderungen
                    \item Definierte \textbf{Schwellenwerte} für Eingriff
                    \item \textbf{Rollback}-Strategie
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

    \frametitle{Key Takeaways}
    \begin{enumerate}
        \item \textbf{ML lernt aus Daten} -- keine explizite Programmierung nötig
        \vspace{0.2cm}
        \item \textbf{Überwacht vs. Unüberwacht:} Mit oder ohne Beispiellösungen
        \vspace{0.2cm}
        \item \textbf{Overfitting vermeiden:} Das Modell soll generalisieren können
        \vspace{0.2cm}
        \item \textbf{Precision \& Recall:} Wichtiger als reine Trefferquote (Accuracy)
        \vspace{0.2cm}
        \item \textbf{Gradient Boosting:} State-of-the-Art für Tabellendaten
        \vspace{0.2cm}
        \item \textbf{Erklärbarkeit \& Governance:} Essenziell für regulierte Bereiche
    \end{enumerate}
    \vspace{0.3cm}
    \begin{block}{Literaturempfehlung}
        Géron, A. (2022). \textit{Hands-On Machine Learning with Scikit-Learn, Keras, and TensorFlow}. O'Reilly. \cite{geron2022hands}
    \end{block}
\end{frame}
