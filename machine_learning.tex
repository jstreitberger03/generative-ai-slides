\section{Grundkonzepte}
%===============================================================================

\begin{frame}
    \frametitle{Was ist Maschinelles Lernen?}
    \begin{columns}
        \begin{column}{0.55\textwidth}
            \begin{block}{Kernidee}
                Statt Regeln manuell zu programmieren, lernt der Computer die Regeln \textbf{aus Daten}.
            \end{block}
            \vspace{0.3cm}
            \textbf{Traditionell:}\\
            Programmierer $\rightarrow$ Regeln $\rightarrow$ Computer
            \vspace{0.2cm}
            
            \textbf{ML:}\\
            Daten + Ergebnisse $\rightarrow$ Computer $\rightarrow$ Regeln
        \end{column}
        \begin{column}{0.42\textwidth}
            \begin{exampleblock}{AML-Beispiel}
                \textbf{Regelbasiert:}\\
                „Melde alle TX $>$ \texteuro 10.000 in Hochrisikoländer"
                \vspace{0.2cm}
                
                \textbf{ML-basiert:}\\
                System erkennt verdächtige Muster aus historischen SAR-Fällen -- auch \textit{unbekannte}.
            \end{exampleblock}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}
    \frametitle{Taxonomie: Überwacht vs. Unüberwacht}
    \begin{center}
        \includegraphics[width=0.85\linewidth]{plots/supervised_vs_unsupervised.pdf}
    \end{center}
    \begin{columns}
        \begin{column}{0.48\textwidth}
            \begin{block}{Überwacht (Supervised)}
                \begin{itemize}
                    \item Training mit \textbf{gelabelten} Daten
                    \item Ziel: Vorhersage für neue Fälle
                    \item \textit{Bsp: Alert-Klassifikation}
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{block}{Unüberwacht (Unsupervised)}
                \begin{itemize}
                    \item Training \textbf{ohne Labels}
                    \item Ziel: Strukturen entdecken
                    \item \textit{Bsp: Kundensegmentierung}
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}
    \frametitle{Taxonomie: Instanzbasiert vs. Modellbasiert}
    \begin{center}
        \includegraphics[width=0.85\linewidth]{plots/instance_vs_model.pdf}
    \end{center}
    \begin{columns}
        \begin{column}{0.48\textwidth}
            \begin{block}{Instanzbasiert}
                \begin{itemize}
                    \item Speichert alle Trainingsdaten
                    \item Vergleicht neue Fälle mit Ähnlichen
                    \item \textit{Bsp: k-Nearest Neighbors}
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{block}{Modellbasiert}
                \begin{itemize}
                    \item Lernt Parameter ($\theta$) aus Daten
                    \item Kompaktes Modell, schnelle Vorhersage
                    \item \textit{Bsp: Logistische Regression}
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}
    \frametitle{Das fundamentale Problem: Generalisierung}
    \begin{center}
        \includegraphics[width=0.95\linewidth]{plots/overfitting_visualization.pdf}
    \end{center}
    \vspace{-0.2cm}
    \begin{alertblock}{Ziel des ML}
        Das Modell soll auf \textbf{neuen, ungesehenen Daten} gut funktionieren -- nicht nur auf den Trainingsdaten!
    \end{alertblock}

    \frametitle{k-Nearest Neighbors: Ähnlichkeitsbasierte Klassifikation}
    \begin{columns}
        \begin{column}{0.48\textwidth}

            \vspace{0.3cm}
            \begin{exampleblock}{AML}
                Ist diese Transaktion ähnlich zu bekannten Betrugsfällen?

\section{Evaluation \& Metriken}
%===============================================================================

\begin{frame}
    \frametitle{Warum Accuracy nicht ausreicht}
    \begin{columns}
        \begin{column}{0.55\textwidth}
            \textbf{Problem: Unbalancierte Daten}
            \begin{itemize}
                \item AML: $<$1\% verdächtige Transaktionen
                \item Ein Modell, das \textit{immer} „normal" sagt, hat 99\% Accuracy!
            \end{itemize}
            \vspace{0.3cm}
            \textbf{Bessere Metriken:}
            \begin{itemize}
                \item \textbf{Precision:} Wie viele Alarme sind echt?
                \item \textbf{Recall:} Wie viele Betrugsfälle gefunden?
                \item \textbf{F1-Score:} Balance beider
            \end{itemize}
        \end{column}
        \begin{column}{0.43\textwidth}
            \begin{alertblock}{AML-Abwägung}
                \textbf{Hoher Recall} = weniger übersehene Fälle\\
                (regulatorisch wichtig!)
                \vspace{0.2cm}
                
                \textbf{Hohe Precision} = weniger Fehlalarme\\
                (Effizienz)
                \vspace{0.2cm}
                
                $\rightarrow$ Schwellenwert je nach Risiko-Appetit wählen
            \end{alertblock}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}
    \frametitle{Konfusionsmatrix \& Precision-Recall}
    \begin{center}
        \includegraphics[width=0.9\linewidth]{plots/precision_recall.pdf}
    \end{center}
    \vspace{-0.2cm}
    \[
        \text{Precision} = \frac{TP}{TP + FP} \qquad
        \text{Recall} = \frac{TP}{TP + FN} \qquad
        F_1 = 2 \, \frac{P \, R}{P + R}
    \]
\end{frame}

\begin{frame}
    \frametitle{Train/Validation/Test: Das Daten-Dreieck}
    \begin{columns}
        \begin{column}{0.55\textwidth}
            \begin{enumerate}
                \item \textbf{Training Set (60-70\%)}\\
                      Modell lernt daraus
                \item \textbf{Validation Set (15-20\%)}\\
                      Hyperparameter-Tuning
                \item \textbf{Test Set (15-20\%)}\\
                      Finale, unberührte Evaluation
            \end{enumerate}
            \vspace{0.3cm}
            \begin{alertblock}{Goldene Regel}
                Das Test Set darf \textbf{niemals} für Modellentscheidungen genutzt werden!
            \end{alertblock}
        \end{column}
        \begin{column}{0.42\textwidth}
            \begin{block}{Cross-Validation}
                Bei wenig Daten: K-Fold CV
                \begin{itemize}
                    \item Teile in $k$ Teile
                    \item Trainiere $k$ mal
                    \item Mittle Ergebnisse
                \end{itemize}
                $\rightarrow$ Robustere Schätzung
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}
    \frametitle{Fragen für Vor-Ort-PrüferInnen}
    \begin{columns}
        \begin{column}{0.48\textwidth}
            \begin{block}{Modellentwicklung}
                \begin{itemize}
                    \item Welche \textbf{Daten} wurden für das Training verwendet?
                    \item Wie wurde \textbf{Overfitting} verhindert?
                    \item Welche \textbf{Features} nutzt das Modell?
                    \item Gibt es eine \textbf{Baseline} zum Vergleich?
                \end{itemize}
            \end{block}
            \vspace{0.2cm}
            \begin{block}{Validierung}
                \begin{itemize}
                    \item Welche \textbf{Metriken} werden verwendet?
                    \item Wie oft wird das Modell \textbf{re-validiert}?
                    \item Gibt es \textbf{Out-of-Time}-Tests?
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{block}{Governance \& Erklärbarkeit}
                \begin{itemize}
                    \item Wer ist für das Modell \textbf{verantwortlich}?
                    \item Ist das Modell \textbf{erklärbar}? (SHAP, Feature Importance)
                    \item Wie werden \textbf{Änderungen} dokumentiert?
                \end{itemize}
            \end{block}
            \vspace{0.2cm}
            \begin{block}{Monitoring \& Drift}
                \begin{itemize}
                    \item Wird \textbf{Model Drift} überwacht?
                    \item Was passiert bei \textbf{Performance-Einbruch}?
                    \item Gibt es \textbf{Fallback}-Mechanismen?
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}
    \frametitle{Red Flags bei ML-Implementierungen}
    \begin{columns}
        \begin{column}{0.48\textwidth}
            \begin{alertblock}{Warnsignale}
                \begin{itemize}
                    \item Keine klare \textbf{Dokumentation}
                    \item Test Set wurde für Tuning genutzt
                    \item Nur \textbf{Accuracy} als Metrik
                    \item Kein \textbf{Monitoring} in Produktion
                    \item „Black Box" ohne Erklärbarkeit
                    \item Keine \textbf{Governance}-Struktur
                    \item Modell seit Jahren nicht \textbf{re-trainiert}
                \end{itemize}
            \end{alertblock}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{block}{Best Practices}
                \begin{itemize}
                    \item Klare \textbf{Model Risk Management} Policy
                    \item Regelmäßige \textbf{Validierung} (jährlich)
                    \item \textbf{Challenger Models} zum Vergleich
                    \item \textbf{Human-in-the-Loop} bei kritischen Entscheidungen
                    \item Dokumentierte \textbf{Feature-Auswahl}
                    \item \textbf{Audit Trail} aller Änderungen
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

\section{Zusammenfassung}

\begin{frame}
    \frametitle{Key Takeaways}
    \begin{enumerate}
        \item \textbf{ML lernt Regeln aus Daten} -- keine explizite Programmierung
        \vspace{0.2cm}
        \item \textbf{Überwacht vs. Unüberwacht:} Mit oder ohne Labels
        \vspace{0.2cm}
        \item \textbf{Generalisierung ist das Ziel} -- Overfitting vermeiden!
        \vspace{0.2cm}
        \item \textbf{Accuracy reicht nicht} -- Precision, Recall, F1 betrachten
        \vspace{0.2cm}
        \item \textbf{Gradient Boosting} ist oft State-of-the-Art für Tabellendaten
        \vspace{0.2cm}
        \item \textbf{Erklärbarkeit \& Governance} sind essenziell für regulierte Bereiche
    \end{enumerate}
